地基0

缓存 / 特征层是干嘛的？

先一句话定位

接入层把请求分到机器，计算层把活干快——但只要一查数据（用户画像、设备指纹、黑白名单、历史行为），就得去存储层（DB）。DB 慢、也扛不住千万 QPS 直接怼。

缓存层 = 在计算层和存储层之间架一层"快速记忆"：把热数据放内存，挡住绝大多数查询，只让极少数漏到 DB。风控里这层尤其重（也叫特征层）：一次风控决策要读几十上百个特征，全靠缓存毫秒级返回。怎么把这层做好？还是那三根轴：

轴② 纵向减负
缓存存在的本质：给慢存储减负

缓存这一层的存在本身就是"纵向减负"——把热数据从慢的存储（磁盘/DB）搬到快的内存，别每次都去慢的地方查。手段：选对读写模式、多级缓存、淘汰策略。

E01

为什么要缓存：内存 vs 磁盘

同样查 20 次，从"磁盘/远程"查 vs 从"内存缓存"查，看耗时差多少。

import java.util.HashMap;
// 缓存层 | 轴② 减负 | 为什么要缓存：内存比"磁盘/远程"快几个数量级
public class E01_WhyCache {
    // 模拟"去数据库/磁盘查"——慢（要走磁盘 IO / 网络）
    static String fromDisk(String key) {
        try { Thread.sleep(50); } catch (InterruptedException e) {}   // 一次磁盘/远程查 ~ 毫秒级
        return "value_of_" + key;
    }
    // 内存里的缓存——快（纳秒级）
    static final HashMap<String, String> mem = new HashMap<>();
    public static void main(String[] a) {
        mem.put("user42", "value_of_user42");
        int N = 20;
        long t0 = System.nanoTime();
        for (int i = 0; i < N; i++) fromDisk("user42");
        long diskMs = (System.nanoTime() - t0) / 1_000_000;
        long t1 = System.nanoTime();
        for (int i = 0; i < N; i++) mem.get("user42");
        long memNs = System.nanoTime() - t1;
        System.out.println("同样查 " + N + " 次：");
        System.out.println("  从磁盘/远程查：" + diskMs + " ms");
        System.out.println("  从内存缓存查：" + (memNs / 1000) + " 微秒(us) = " + (memNs / 1_000_000.0) + " ms");
        System.out.println("  快了约 " + (diskMs * 1_000_000 / Math.max(memNs, 1)) + " 倍");
        System.out.println(">> 内存比磁盘/远程快成千上万倍。缓存的本质就是'把热数据放内存，别每次都去慢的地方查'。");
    }
}

这说明了什么

内存读是纳秒级，磁盘/远程读是毫秒级，差了几万倍。缓存的本质就是：把热数据放内存，别每次都去慢的地方查。风控里一次决策要读几十上百个特征（用户画像、设备指纹、名单），全靠缓存才能毫秒级返回——所以这层也叫"特征层"。

E02

Cache-Aside：最常用的读写模式

读：先查缓存，没有再回源 DB 并回填。写：更新 DB，然后删缓存。

import java.util.*;
import java.util.concurrent.atomic.AtomicInteger;
// 缓存层 | 轴② 减负 | Cache-Aside：最常用的读写模式
public class E02_CacheAside {
    static final Map<String, String> cache = new HashMap<>();
    static final Map<String, String> db = new HashMap<>(Map.of("user42", "余额100"));
    static final AtomicInteger dbReads = new AtomicInteger();

    static String read(String key) {                 // 读：先查缓存，没有再回源 DB 并回填
        String v = cache.get(key);
        if (v != null) { System.out.println("  读 " + key + " → 命中缓存，直接返回"); return v; }
        dbReads.incrementAndGet();
        v = db.get(key);
        System.out.println("  读 " + key + " → 缓存没有，回源 DB(第" + dbReads.get() + "次)，并回填缓存");
        cache.put(key, v);
        return v;
    }
    static void write(String key, String val) {      // 写：更新 DB，然后删缓存（不是改缓存）
        db.put(key, val);
        cache.remove(key);
        System.out.println("  写 " + key + "=" + val + " → 更新 DB + 删缓存");
    }
    public static void main(String[] a) {
        System.out.println("连续读两次同一个 key：");
        read("user42"); read("user42");
        System.out.println("更新它：");
        write("user42", "余额200");
        System.out.println("再读两次：");
        read("user42"); read("user42");
        System.out.println(">> Cache-Aside：读 miss 才回源并回填；写时'删'缓存让下次自然回源最新值。全程回源 DB 仅 " + dbReads.get() + " 次。");
    }
}
运行中… javac E02_CacheAside.java && java E02_CacheAside
▌ 真实输出（本机 10 核 · OpenJDK 21 实跑）
连续读两次同一个 key：

  读 user42 → 命中缓存，直接返回

>> Cache-Aside：读 miss 才回源并回填；写时'删'缓存让下次自然回源最新值。全程回源 DB 仅 2 次。
Cache-Aside 旁路缓存

为什么写时是"删缓存"而不是"改缓存"？因为并发下两个写同时"改"缓存，可能把旧值盖在新值上；"删"让下次读自然回源最新值，更安全。另一类模式是 Read/Write-Through（由缓存组件自己回源、应用不直接碰 DB）。90% 的业务用的就是 Cache-Aside。

E03

多级缓存：本地 L1 + 分布式 L2

先查最快的本地缓存，没有再查共享的 Redis，最后才回源 DB。

import java.util.*;
import java.util.concurrent.atomic.AtomicInteger;
// 缓存层 | 轴② 减负 | 多级缓存：本地(L1) + 分布式(L2)
public class E03_MultiLevel {
    static final Map<String, String> L1 = new HashMap<>();   // 本地缓存(进程内, 最快, 如 Caffeine)
    static final Map<String, String> L2 = new HashMap<>();   // 分布式缓存(跨机器共享, 如 Redis)
    static final Map<String, String> db = new HashMap<>(Map.of("u", "特征值"));
    static final AtomicInteger dbHit = new AtomicInteger();

    static String read(String key) {
        String v = L1.get(key);
        if (v != null) { System.out.println("  " + key + " → L1 本地命中(最快)"); return v; }
        v = L2.get(key);
        if (v != null) { L1.put(key, v); System.out.println("  " + key + " → L1 没有, L2 命中, 回填 L1"); return v; }
        dbHit.incrementAndGet(); v = db.get(key); L2.put(key, v); L1.put(key, v);
        System.out.println("  " + key + " → L1/L2 都没有, 回源 DB, 回填 L2+L1");
        return v;
    }
    public static void main(String[] a) {
        read("u");                                           // 全 miss → DB
        read("u");                                           // L1 命中
        L1.clear();
        System.out.println("(假设本地缓存过期 / 请求换到了另一台机器, L1 清空)");
        read("u");                                           // L1 miss, L2 命中
        System.out.println(">> 多级缓存：先查最快的本地 L1, 没有再查共享的 L2(Redis), 都没有才回源 DB。全程回源 DB 仅 " + dbHit.get() + " 次。");
    }
}
运行中… javac E03_MultiLevel.java && java E03_MultiLevel
▌ 真实输出（本机 10 核 · OpenJDK 21 实跑）
  u → L1/L2 都没有, 回源 DB, 回填 L2+L1
  u → L1 本地命中(最快)
(假设本地缓存过期 / 请求换到了另一台机器, L1 清空)
  u → L1 没有, L2 命中, 回填 L1
>> 多级缓存：先查最快的本地 L1, 没有再查共享的 L2(Redis), 都没有才回源 DB。全程回源 DB 仅 1 次。
多级缓存 L1 / L2

L1 本地缓存（Caffeine，进程内，纳秒级，但每台机器各存一份、有一致性问题）+ L2 分布式缓存（Redis，跨机器共享，微秒级）。风控决策机把热名单放本地 L1，绝大多数查询本地就挡住了，连 Redis 都不用打。代价是 L1 的更新延迟——通常给 L1 设很短的 TTL 容忍短暂不一致。

E04

LRU 淘汰：内存满了删谁

缓存容量有限，满了必须淘汰。LRU = 删"最久没被访问"的那个。

import java.util.*;
// 缓存层 | 轴② 减负 | LRU 淘汰：内存满了, 删"最久没用"的
public class E04_LRU {
    // LinkedHashMap accessOrder=true：每次访问把元素挪到末尾；超容量就删最前面那个(最久没用)
    static class LRU<K, V> extends LinkedHashMap<K, V> {
        final int cap;
        LRU(int cap) { super(16, 0.75f, true); this.cap = cap; }
        protected boolean removeEldestEntry(Map.Entry<K, V> e) {
            boolean full = size() > cap;
            if (full) System.out.println("    容量满(" + cap + "), 淘汰最久没用的: " + e.getKey());
            return full;
        }
    }
    public static void main(String[] a) {
        LRU<String, Integer> cache = new LRU<>(3);
        System.out.println("容量=3, 依次放入 A B C：");
        cache.put("A", 1); cache.put("B", 2); cache.put("C", 3);
        System.out.println("  现在缓存: " + cache.keySet());
        System.out.println("访问一下 A（A 变成最近用过的）：");
        cache.get("A");
        System.out.println("  现在顺序: " + cache.keySet() + "   (B 成了最久没用的)");
        System.out.println("放入 D（超容量, 触发淘汰）：");
        cache.put("D", 4);
        System.out.println("  现在缓存: " + cache.keySet() + "   (B 被淘汰, 因为它最久没被访问)");
        System.out.println(">> LRU = Least Recently Used，内存满时优先淘汰最久没访问的。Redis、Caffeine 都支持。");
    }
}
运行中… javac E04_LRU.java && java E04_LRU
▌ 真实输出（本机 10 核 · OpenJDK 21 实跑）
容量=3, 依次放入 A B C：
  现在缓存: [A, B, C]
访问一下 A（A 变成最近用过的）：
  现在顺序: [B, C, A]   (B 成了最久没用的)
放入 D（超容量, 触发淘汰）：
    容量满(3), 淘汰最久没用的: B
  现在缓存: [C, A, D]   (B 被淘汰, 因为它最久没被访问)
>> LRU = Least Recently Used，内存满时优先淘汰最久没访问的。Redis、Caffeine 都支持。
关键点

LinkedHashMap 构造时传 accessOrder=true + 重写 removeEldestEntry，就是手写 LRU 的标准技巧（面试高频手写题）。除 LRU 外还有 LFU（按访问频率）、TTL（按过期时间）、FIFO。Redis 的 maxmemory-policy 可配这些策略。

轴③ 横切兜底
缓存的四大翻车现场

缓存一旦没扛住，压力就直接砸到后面的 DB。这一轴专治缓存的四个经典翻车点：穿透（查不存在的）、击穿（一个热点 key 过期）、雪崩（一大片 key 同时过期/集群挂）、一致性（缓存和 DB 对不上）。

E05

缓存穿透 + 缓存空值

狂查一个"根本不存在"的 key，缓存永远 miss，每次都直击 DB。

import java.util.*;
import java.util.concurrent.atomic.AtomicInteger;
// 缓存层 | 轴③ 兜底 | 缓存穿透：狂查"根本不存在的 key", 每次都 miss 直击 DB
public class E05_Penetration {
    static final Map<String, String> db = new HashMap<>();        // DB 里根本没有 'ghost' 这个 key

    // ① 无防护：null 不回填 → 每次都得查 DB
    static final Map<String, String> cache = new HashMap<>();
    static final AtomicInteger dbReads = new AtomicInteger();
    static String readNaive(String key) {
        String v = cache.get(key);
        if (v != null) return v;
        dbReads.incrementAndGet();
        v = db.get(key);                                          // null! DB 里没有
        if (v != null) cache.put(key, v);                        // null 不回填 → 下次还得查 DB
        return v;
    }

    // ② 缓存空值：即使不存在也缓存一个"空"，挡住后续请求
    static final Map<String, String> cache2 = new HashMap<>();
    static final AtomicInteger dbReads2 = new AtomicInteger();
    static String readGuarded(String key) {
        if (cache2.containsKey(key)) return cache2.get(key);     // 即使值是空, 也算命中
        dbReads2.incrementAndGet();
        String v = db.get(key);
        cache2.put(key, v == null ? "" : v);                     // 不存在也缓存一个空值(可设短TTL)
        return v;
    }
    public static void main(String[] a) {
        System.out.println("狂查一个不存在的 key 'ghost' 共 5 次：");
        for (int i = 0; i < 5; i++) readNaive("ghost");
        System.out.println("  【无防护】  打到 DB 次数: " + dbReads.get() + "   (5 次全穿透到 DB!)");
        for (int i = 0; i < 5; i++) readGuarded("ghost");
        System.out.println("  【缓存空值】打到 DB 次数: " + dbReads2.get() + "   (第 1 次后就被空值挡住)");
        System.out.println(">> 穿透 = 查不存在的数据, 绕过缓存直击 DB。解法: 缓存空值(简单) 或 布隆过滤器(省内存)。");
    }
}
运行中… javac E05_Penetration.java && java E05_Penetration
▌ 真实输出（本机 10 核 · OpenJDK 21 实跑）
狂查一个不存在的 key 'ghost' 共 5 次：
  【无防护】  打到 DB 次数: 5   (5 次全穿透到 DB!)
  【缓存空值】打到 DB 次数: 1   (第 1 次后就被空值挡住)
>> 穿透 = 查不存在的数据, 绕过缓存直击 DB。解法: 缓存空值(简单) 或 布隆过滤器(省内存)。
缓存穿透 Penetration

查的是 DB 里也没有的数据（常见于恶意攻击：拿一堆不存在的 ID 狂打），缓存永远命不中，每次都穿到 DB。解法：① 缓存空值（把"不存在"也缓存起来，简单，但占内存、要设短 TTL）② 布隆过滤器（用极小内存 O(1) 判断"这个 key 一定不存在"，存在则放行，省内存）。

E06

缓存击穿 + 互斥锁

单个热点 key 过期的一瞬间，100 个并发请求同时 miss、一起回源打爆 DB。

import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;
// 缓存层 | 轴③ 兜底 | 缓存击穿：单个热点 key 过期瞬间, 大量请求同时回源打爆 DB
public class E06_Breakdown {
    static String loadFromDB(AtomicInteger counter) {
        counter.incrementAndGet();
        try { Thread.sleep(100); } catch (InterruptedException e) {}   // 回源耗时 100ms
        return "hot_value";
    }

    // ① 无锁：谁发现 null 谁就回源 → 100 个线程同时进来都回源
    static volatile String cache = null;
    static final AtomicInteger dbReads = new AtomicInteger();
    static String readNaive() {
        if (cache != null) return cache;
        cache = loadFromDB(dbReads);
        return cache;
    }

    // ② 互斥锁：只放一个线程回源, 其余等它回填(双重检查)
    static volatile String cache2 = null;
    static final AtomicInteger dbReads2 = new AtomicInteger();
    static final Semaphore lock = new Semaphore(1);
    static String readLocked() {
        if (cache2 != null) return cache2;
        lock.acquireUninterruptibly();
        try {
            if (cache2 != null) return cache2;                   // 拿到锁再看一眼, 可能别人已回填
            cache2 = loadFromDB(dbReads2);
            return cache2;
        } finally { lock.release(); }
    }

    static void blast(Runnable r) throws Exception {
        ExecutorService p = Executors.newFixedThreadPool(100);
        CountDownLatch l = new CountDownLatch(100);
        for (int i = 0; i < 100; i++) p.submit(() -> { r.run(); l.countDown(); });
        l.await(); p.shutdown();
    }
    public static void main(String[] a) throws Exception {
        System.out.println("热点 key 刚过期(缓存为空), 100 个请求同时涌入：");
        blast(E06_Breakdown::readNaive);
        System.out.println("  【无锁】    回源 DB 次数: " + dbReads.get() + "   (一大批全打到 DB!)");
        blast(E06_Breakdown::readLocked);
        System.out.println("  【互斥锁】  回源 DB 次数: " + dbReads2.get() + "   (只有 1 个回源, 其余等回填)");
        System.out.println(">> 击穿 = 单个热点 key 过期瞬间被并发击穿。解法: 互斥锁只放一个回源, 或'逻辑过期'不真删。");
    }
}
运行中… javac E06_Breakdown.java && java E06_Breakdown
▌ 真实输出（本机 10 核 · OpenJDK 21 实跑）
热点 key 刚过期(缓存为空), 100 个请求同时涌入：
  【无锁】    回源 DB 次数: 100   (一大批全打到 DB!)
  【互斥锁】  回源 DB 次数: 1   (只有 1 个回源, 其余等回填)
>> 击穿 = 单个热点 key 过期瞬间被并发击穿。解法: 互斥锁只放一个回源, 或'逻辑过期'不真删。
缓存击穿 Breakdown

某个热点 key（首页、爆款商品）过期的一刹那，海量并发同时发现缓存空了、一起回源，DB 瞬间被打爆。和"穿透"区分：击穿的 key 本来有、只是刚过期；穿透的 key 根本不存在。解法：互斥锁（只放一个线程回源、其余等回填，见输出 100→1）或逻辑过期（不真删，后台异步刷新）。

E07

缓存雪崩 + TTL 打散

大量 key 设了相同 TTL → 同一时刻一起过期 → 瞬时回源洪峰打爆 DB。

import java.util.*;
// 缓存层 | 轴③ 兜底 | 缓存雪崩：大量 key 同一时刻过期, 瞬间全部回源打爆 DB
public class E07_Avalanche {
    public static void main(String[] a) {
        int N = 1000;
        Random r = new Random(42);

        // ① 所有 key 设同样的 TTL → 同一时刻一起过期
        //    => 1000 个 key 全落在同一个 100ms 窗口, 瞬时回源 QPS = 1000/0.1s = 10000
        int sameWindowPeak = N;

        // ② TTL 加随机抖动: base + rand(0..1000ms) → 散到 1 秒内的 10 个窗口
        int[] bucket = new int[10];                        // 每格 = 100ms 窗口内过期的 key 数
        for (int i = 0; i < N; i++) {
            int b = (int) (r.nextDouble() * 10);           // 落在 0..9 号窗口
            bucket[b]++;
        }
        int jitterPeak = Arrays.stream(bucket).max().getAsInt();

        System.out.println("1000 个 key 的过期时间分布（每格 = 一个 100ms 窗口内过期的 key 数）：");
        System.out.println("  【同一 TTL】   [" + sameWindowPeak + ", 0, 0, 0, 0, 0, 0, 0, 0, 0]");
        System.out.println("                 → 同一瞬间 " + N + " 个全过期, 峰值回源 QPS ≈ " + (sameWindowPeak * 10) + ", DB 被打爆");
        System.out.println("  【加随机抖动】 " + Arrays.toString(bucket));
        System.out.println("                 → 摊到 1 秒内, 单窗口最多 " + jitterPeak + " 个, 峰值回源 QPS ≈ " + (jitterPeak * 10) + "（降了约 " + (sameWindowPeak / jitterPeak) + " 倍）");
        System.out.println(">> 雪崩 = 大量 key 同时过期(或缓存集群整挂)。解法: TTL 加随机抖动打散 + 缓存高可用 + 限流降级。");
    }
}

两种雪崩，两种解法

① 大量 key 同时过期：给 TTL 加随机抖动打散（见输出，峰值 QPS 降了约 8 倍）。② 整个缓存集群挂了：靠 Redis 高可用 + 本地缓存兜底 + 限流降级（fail-open）。和"击穿"区分：击穿是"一个"热点 key，雪崩是"一大片"key 同时出事。

E08

缓存一致性 + 延迟双删

并发下，读线程的"旧值回填"会和写线程抢顺序，导致缓存存了旧值。看真实 race。

import java.util.*;
import java.util.concurrent.ConcurrentHashMap;
// 缓存层 | 轴③ 兜底 | 缓存一致性：并发下缓存可能存到旧值 + 延迟双删
public class E08_Consistency {
    static Map<String, String> cache = new ConcurrentHashMap<>();
    static Map<String, String> db = new ConcurrentHashMap<>();
    static void sleep(long ms) { try { Thread.sleep(ms); } catch (InterruptedException e) {} }

    // 问题：缓存已失效。读线程回源拿到旧值, 还没回填时, 写线程更新了 DB 并删了缓存,
    //       读线程随后把"旧值"写回缓存 → 缓存脏了。
    static void demoRace() throws Exception {
        cache.clear(); db.put("k", "OLD");
        System.out.println("初始: DB=OLD, 缓存空");
        Thread reader = new Thread(() -> {
            String v = db.get("k");                       // t≈0  回源读到旧值 OLD
            System.out.println("  [读线程] 回源读到: " + v);
            sleep(100);                                   // 读线程慢了一下
            cache.put("k", v);                            // t≈100 把 OLD 写回缓存
            System.out.println("  [读线程] 回填缓存: " + v);
        });
        Thread writer = new Thread(() -> {
            sleep(50);                                    // t≈50 写线程介入
            db.put("k", "NEW");                           // 更新 DB
            cache.remove("k");                            // 删缓存
            System.out.println("  [写线程] 更新 DB=NEW, 删缓存");
        });
        reader.start(); writer.start(); reader.join(); writer.join();
        String c = cache.get("k");
        System.out.println("结果: DB=" + db.get("k") + ", 缓存=" + c + "  → " + ("NEW".equals(c) ? "一致" : "不一致! 缓存是旧值 OLD") + "\n");
    }

    // 解法：延迟双删 —— 写时先删缓存 → 更新 DB → 等一会儿(超过读线程回填耗时) → 再删一次
    static void demoDoubleDelete() throws Exception {
        cache.clear(); db.put("k", "OLD");
        System.out.println("初始: DB=OLD, 缓存空 (延迟双删版)");
        Thread reader = new Thread(() -> {
            String v = db.get("k");
            System.out.println("  [读线程] 回源读到: " + v);
            sleep(100); cache.put("k", v);
            System.out.println("  [读线程] 回填缓存: " + v);
        });
        Thread writer = new Thread(() -> {
            sleep(50);
            cache.remove("k");                            // 删 1
            db.put("k", "NEW");                           // 更新 DB
            System.out.println("  [写线程] 删缓存 + 更新 DB=NEW, 安排延迟再删");
            sleep(200);                                   // 等超过读线程的回填(100ms)
            cache.remove("k");                            // 删 2: 把读线程刚回填的旧值再清掉
            System.out.println("  [写线程] 延迟再删缓存(清掉可能被回填的旧值)");
        });
        reader.start(); writer.start(); reader.join(); writer.join();
        String c = cache.get("k");
        System.out.println("结果: DB=" + db.get("k") + ", 缓存=" + (c == null ? "空(下次读自然回源 NEW)" : c) + "  → 一致");
    }
    public static void main(String[] a) throws Exception {
        System.out.println("=== 问题: 普通'更DB+删缓存'在并发下出脏数据 ===");
        demoRace();
        System.out.println("=== 解法: 延迟双删 ===");
        demoDoubleDelete();
        System.out.println(">> 一致性核心是'写操作'怎么和'读回填'抢顺序。延迟双删 = 删→更DB→延迟再删, 兜住读线程的旧回填。");
    }
}
运行中… javac E08_Consistency.java && java E08_Consistency
▌ 真实输出（本机 10 核 · OpenJDK 21 实跑）
=== 问题: 普通'更DB+删缓存'在并发下出脏数据 ===
初始: DB=OLD, 缓存空
  [读线程] 回源读到: OLD
  [写线程] 更新 DB=NEW, 删缓存
  [读线程] 回填缓存: OLD

>> 一致性核心是'写操作'怎么和'读回填'抢顺序。延迟双删 = 删→更DB→延迟再删, 兜住读线程的旧回填。
缓存一致性 Consistency

这是经典难题。即使用 Cache-Aside 的"更 DB + 删缓存"，并发下读线程仍可能把旧值回填进去（本 demo 真实复现）。缓解：延迟双删（删→更DB→延迟再删一次，兜住读线程的旧回填）、给缓存设 TTL 兜底（最终一致）、订阅 binlog（canal 监听 DB 变更异步删缓存）。强一致几乎做不到，工程上普遍接受"最终一致"。

轴① 横向摊薄
单机放不下就分片

单机内存有限，热数据放不下，就把缓存分片到多台（缓存集群）。难点是扩容/缩容时怎么不让 key 大规模迁移、引发雪崩。

E09

一致性哈希分片

单机内存放不下，把缓存分片到多台。加节点时，怎么只迁移少量 key？

import java.util.*;
// 缓存层 | 轴① 摊薄 | 一致性哈希分片：加节点只迁移少量 key（呼应接入层 L5）
public class E09_ConsistentHash {
    static TreeMap<Integer, String> ring = new TreeMap<>();
    static final int VNODES = 100;                          // 每个物理节点放 100 个虚拟节点, 让分布均匀
    // 哈希函数：String.hashCode() 对相似输入会聚成一团, 这里加 murmur 雪崩混淆打散到整个范围
    static int hash(String s) {
        int h = s.hashCode();
        h ^= (h >>> 16); h *= 0x85ebca6b; h ^= (h >>> 13); h *= 0xc2b2ae35; h ^= (h >>> 16);
        return h & 0x7fffffff;
    }
    static void addNode(String node) {
        for (int i = 0; i < VNODES; i++) ring.put(hash(node + "#" + i), node);
    }
    static String locate(String key) {                      // 顺时针找环上第一个节点
        Map.Entry<Integer, String> e = ring.ceilingEntry(hash(key));
        if (e == null) e = ring.firstEntry();               // 环形: 超过最大就回到第一个
        return e.getValue();
    }
    public static void main(String[] a) {
        for (String n : List.of("缓存A", "缓存B", "缓存C")) addNode(n);
        int N = 10000;
        Map<String, String> before = new HashMap<>();
        for (int i = 0; i < N; i++) { String k = "key" + i; before.put(k, locate(k)); }
        Map<String, Long> dist = new TreeMap<>();
        before.forEach((k, v) -> dist.merge(v, 1L, Long::sum));
        System.out.println("3 个缓存节点, " + N + " 个 key 的分布: " + dist);

        addNode("缓存D");                                    // 加第 4 个节点
        int moved = 0;
        for (int i = 0; i < N; i++) { String k = "key" + i; if (!locate(k).equals(before.get(k))) moved++; }
        System.out.println("加入第 4 个节点 '缓存D' 后:");
        System.out.println("  迁移的 key 数: " + moved + "/" + N + " (" + (moved * 100 / N) + "%)");
        System.out.println("  对比: 若用'取模 hash%N'分片, 从 3 台变 4 台几乎所有 key 都要迁移(约 " + (100 - 100 / 4) + "%)");
        System.out.println(">> 一致性哈希: 加/减节点只影响环上相邻的一小段 key, 大部分 key 不动 → 扩容不雪崩。");
    }
}

一致性哈希 vs 取模分片

取模 hash % N 分片，从 3 台扩到 4 台要迁移约 75% 的 key（迁移即大面积 miss → 雪崩）；一致性哈希加节点只动环上相邻一小段，只迁移约 26%（≈ 1/新节点数）。虚拟节点（每个物理节点放 100 个）解决分布不均。这正是接入层 L5 讲过的"同用户同机器"那个一致性哈希；Redis Cluster 用的是 16384 个哈希槽，思想一致。

补一个高频考点：热点 key（热 key）

一致性哈希解决了"整体分布均匀"，但挡不住单个超热 key（如明星塌房、爆款秒杀）把流量集中砸到某一个分片，那个分片单点过载。解法：① 给热 key 加随机后缀拆成多份（key#1…key#N）多副本打散到不同分片；② 在计算层用本地缓存把热 key 兜在进程内，根本不打到缓存集群。

一句话总结缓存层
