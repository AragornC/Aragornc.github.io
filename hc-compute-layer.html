总吞吐 = 单机吞吐 × 机器数。这一轴负责把"机器数"做上去——但前提是无状态：任意请求能落到任意机器、结果都一样，才能随便加机器。

D01

反例：本地状态，一加机器就乱

先看为什么"有状态"的服务不能水平扩展。两台机器各存各的计数。

// 轴① 横向摊薄 | 反例：本地状态 → 多实例数据不一致
public class D01_StatefulBug {
    // 模拟"一台机器"：它把计数存在自己的内存里（本地状态）
    static class Instance {
        private int localCount = 0;        // ← 本地变量，每台机器各存各的
        void handle() { localCount++; }
        int read()    { return localCount; }
    }
    public static void main(String[] args) {
        Instance m1 = new Instance();      // 机器A
        Instance m2 = new Instance();      // 机器B
        Instance[] lb = { m1, m2 };        // 负载均衡轮流把请求分给两台
        for (int i = 0; i < 6; i++) lb[i % 2].handle();   // 6 个 "+1" 请求
        System.out.println("一共发了 6 个 +1 请求");
        System.out.println("机器A 自己数到：" + m1.read());
        System.out.println("机器B 自己数到：" + m2.read());
        System.out.println("用户查询时落到机器A，看到的是：" + m1.read() + "（不是 6！）");
        System.out.println(">> 本地状态让每台机器只知道自己那部分，一加机器就数据不一致。");
    }
}

这说明了什么

计数器存在本地内存里，每台机器只知道自己处理过的那部分。负载均衡一分流，数据就对不上。登录态、计数、累计值这类"状态"放本地，就堵死了加机器这条路。

D02

正解：无状态 + 状态外置

把状态挪到外部存储（这里用 AtomicInteger 模拟 Redis），机器本身不存任何东西。

import java.util.concurrent.atomic.AtomicInteger;
// 轴① 横向摊薄 | 正解：无状态 + 状态外置 → 任意机器结果一致
public class D02_Stateless {
    // 外部共享存储（想象成 Redis），所有机器都读写它
    static final AtomicInteger STORE = new AtomicInteger(0);
    // 机器本身不存任何状态，只是把请求转给外部存储
    static class Instance {
        void handle() { STORE.incrementAndGet(); }   // 状态在外面，不在本地
        int read()    { return STORE.get(); }
    }
    public static void main(String[] args) {
        Instance[] lb = { new Instance(), new Instance(), new Instance() }; // 3 台
        for (int i = 0; i < 6; i++) lb[i % 3].handle();    // 6 个请求随便分
        System.out.println("一共发了 6 个 +1 请求，分给 3 台无状态机器");
        for (int i = 0; i < 3; i++)
            System.out.println("从机器" + i + " 查询，都看到：" + lb[i].read());
        System.out.println(">> 状态外置后，落到哪台机器结果都一样 → 可以随便加机器水平扩展。");
    }
}

无状态 = 水平扩展的前提

机器不存状态，只是"转发器"：状态统一放外部（Redis）。于是请求落到哪台机器结果都一样，就能随便加机器、任意一台挂了也不丢数据。会话、限流计数、去重集合都该这样外置。

D03

串行 vs 线程池

8 个各耗时 100ms 的任务：一个一个串行干，对比用 4 条线程的线程池并发干。

import java.util.concurrent.*;
// 轴② 纵向减负 | 串行 vs 线程池：并发受控地榨干单机
public class D03_PoolVsSerial {
    static void work(int id) {
        try { Thread.sleep(100); } catch (InterruptedException e) {}   // 模拟一次耗时处理
        System.out.println("  任务" + id + " 完成 @ " + Thread.currentThread().getName());
    }
    public static void main(String[] args) throws Exception {
        int N = 8;
        // ① 串行：一个一个来，全在 main 线程上跑
        long t0 = System.currentTimeMillis();
        for (int i = 1; i <= N; i++) work(i);
        System.out.println("串行处理 " + N + " 个任务，耗时 " + (System.currentTimeMillis() - t0) + " ms\n");

        // ② 线程池：只开 4 条线程，复用着把 8 个活干完
        ExecutorService pool = Executors.newFixedThreadPool(4);
        CountDownLatch latch = new CountDownLatch(N);
        long t1 = System.currentTimeMillis();
        for (int i = 1; i <= N; i++) { int id = i; pool.submit(() -> { work(id); latch.countDown(); }); }
        latch.await();
        System.out.println("线程池(4线程)处理 " + N + " 个任务，耗时 " + (System.currentTimeMillis() - t1) + " ms");
        System.out.println(">> 看上面线程名：只有 4 条线程在轮流复用，不是开 8 条。串行≈800ms → 并行≈200ms。");
        pool.shutdown();
    }
}

D04

线程数怎么定？CPU密集 vs IO密集

同样 40 个任务，线程开多少合适？两种场景结论完全相反。

import java.util.concurrent.*;
// 轴② 纵向减负 | 线程数怎么定？CPU密集 vs IO密集，结论相反
public class D04_CpuVsIo {
    static long cpuTask() {                       // 纯计算：把 CPU 吃满
        long s = 0;
        for (int i = 0; i < 60_000_000; i++) s += i % 7;
        return s;
    }
    static void ioTask() {                         // 模拟 IO：线程在"等"，不吃 CPU
        try { Thread.sleep(100); } catch (InterruptedException e) {}
    }
    static long run(int threads, int tasks, boolean cpu) throws Exception {
        ExecutorService p = Executors.newFixedThreadPool(threads);
        CountDownLatch l = new CountDownLatch(tasks);
        long t = System.currentTimeMillis();
        for (int i = 0; i < tasks; i++) p.submit(() -> { if (cpu) cpuTask(); else ioTask(); l.countDown(); });
        l.await(); p.shutdown();
        return System.currentTimeMillis() - t;
    }
    public static void main(String[] a) throws Exception {
        int cores = Runtime.getRuntime().availableProcessors();
        System.out.println("本机核数 = " + cores + "\n");
        System.out.println("【CPU密集】40 个纯计算任务：");
        System.out.println("  " + cores + " 线程(=核数)：" + run(cores, 40, true) + " ms");
        System.out.println("  50 线程(远超核数)：" + run(50, 40, true) + " ms");
        System.out.println("  >> CPU 就那么多核，线程加到核数附近就到顶，再加基本持平、不会线性变快。\n");
        System.out.println("【IO密集】40 个等待任务(各 sleep 100ms)：");
        System.out.println("  " + cores + " 线程：" + run(cores, 40, false) + " ms");
        System.out.println("  40 线程：" + run(40, 40, false) + " ms");
        System.out.println("  >> IO 时线程大多在等、不占 CPU，开多线程能同时等 → 快得多。");
        System.out.println(">> 经验公式：线程数 ≈ 核数 ×(1 + 等待时间/计算时间)。CPU密集≈核数，IO密集可远大于核数。");
    }
}

读懂这组真实数据

CPU密集（纯计算）：10 线程 vs 50 线程基本持平——CPU 就 10 个核，活全在烧 CPU，开再多线程也只能 10 个核轮流跑，多出来的线程只是排队+切换。
IO密集（sleep 等待）：40 线程比 10 线程快约 4 倍——线程大多在"等"、不占 CPU，开更多线程就能同时等。
经验公式：线程数 ≈ 核数 ×(1 + 等待时间/计算时间)。CPU密集≈核数；IO密集可远大于核数。

D05

异步编排：并行调下游

风控要同时查 3 个互不依赖的下游（各 200ms）。串行 vs CompletableFuture 并行。

import java.util.concurrent.*;
// 轴② 纵向减负 | 异步编排：互不依赖的下游并行调，总耗时 sum → max
public class D05_CompletableFuture {
    static String call(String name, int ms) {
        try { Thread.sleep(ms); } catch (InterruptedException e) {}   // 模拟一次下游 RPC
        return name + "结果";
    }
    public static void main(String[] a) throws Exception {
        // ① 串行：一个调完再调下一个
        long t0 = System.currentTimeMillis();
        String r1 = call("风控名单", 200);
        String r2 = call("设备指纹", 200);
        String r3 = call("行为模型", 200);
        System.out.println("串行调用 3 个下游：" + (System.currentTimeMillis() - t0) + " ms  [" + r1 + "," + r2 + "," + r3 + "]");

        // ② 并行：三个同时发出去，只等最慢的那个
        long t1 = System.currentTimeMillis();
        CompletableFuture<String> f1 = CompletableFuture.supplyAsync(() -> call("风控名单", 200));
        CompletableFuture<String> f2 = CompletableFuture.supplyAsync(() -> call("设备指纹", 200));
        CompletableFuture<String> f3 = CompletableFuture.supplyAsync(() -> call("行为模型", 200));
        CompletableFuture.allOf(f1, f2, f3).join();   // 等三个都好
        System.out.println("并行调用 3 个下游：" + (System.currentTimeMillis() - t1) + " ms  [" + f1.get() + "," + f2.get() + "," + f3.get() + "]");
        System.out.println(">> 三个下游互不依赖时，并行把总耗时从 sum(600) 降到 max(200)。");
    }
}
运行中… javac D05_CompletableFuture.java && java D05_CompletableFuture
▌ 真实输出（本机 10 核 · OpenJDK 21 实跑）
串行调用 3 个下游：622 ms  [风控名单结果,设备指纹结果,行为模型结果]
并行调用 3 个下游：211 ms  [风控名单结果,设备指纹结果,行为模型结果]

类比

三个下游像三个查询窗口。串行=排队一个个问，三个加起来 600ms；并行=同时把三个请求发出去，只等最慢的那个，200ms。互不依赖的调用就该并行——总耗时从 sum 降到 max。

D06

虚拟线程：百万级毫无压力

JDK 21 的虚拟线程。一口气开 100 万个"线程"各 sleep 100ms。

import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;
// 轴② 纵向减负 | 虚拟线程(JDK21)：百万级"线程"毫无压力（呼应协程）
public class D06_VirtualThreads {
    public static void main(String[] a) throws Exception {
        int N = 1_000_000;
        AtomicInteger done = new AtomicInteger();
        long t = System.currentTimeMillis();
        // 每个任务一条虚拟线程；try-with-resources 的 close() 会等所有任务跑完
        try (var ex = Executors.newVirtualThreadPerTaskExecutor()) {
            for (int i = 0; i < N; i++) ex.submit(() -> {
                try { Thread.sleep(100); } catch (InterruptedException e) {}   // 模拟 IO 等待
                done.incrementAndGet();
            });
        }
        System.out.println("启动并跑完 " + N + " 个虚拟线程，每个 sleep 100ms");
        System.out.println("完成数 = " + done.get() + "，总耗时 " + (System.currentTimeMillis() - t) + " ms");
        System.out.println(">> 一百万个'线程'毫无压力；换平台线程开 1 万个就可能 OOM。");
        System.out.println(">> 因为虚拟线程在 sleep/IO 等待时会让出底层载体线程，不真占 OS 线程。");
    }
}
运行中… javac D06_VirtualThreads.java && java D06_VirtualThreads
▌ 真实输出（本机 10 核 · OpenJDK 21 实跑）

完成数 = 1000000，总耗时 4638 ms

虚拟线程 Virtual Thread / JDK21

由 JVM 自己调度的超轻量线程。当它 sleep/等 IO 时，会让出底下真正的 OS 线程（载体线程）去干别的活，所以不真占 OS 线程——百万个也扛得住。换成传统平台线程开 1 万个就可能 OOM。这正是你之前学的协程思想的 Java 落地（详见 协程·从0到Java实现）。

D07

本地缓存：算一次，存下来

把"很贵的计算"结果缓存在本地。第一次真算，第二次直接拿。

import java.util.concurrent.ConcurrentHashMap;
// 轴② 纵向减负 | 本地缓存：算过的结果存下来，第二次直接拿
public class D07_LocalCache {
    static final ConcurrentHashMap<String, Integer> cache = new ConcurrentHashMap<>();
    static int expensive(String key) {              // 假装很贵的计算（查库 / 跑模型）
        try { Thread.sleep(300); } catch (InterruptedException e) {}
        return key.length() * 100;
    }
    static int get(String key) {
        // 缓存里没有才真去算，有就直接返回（原子的"没有就计算并放入"）
        return cache.computeIfAbsent(key, D07_LocalCache::expensive);
    }
    public static void main(String[] a) {
        long t0 = System.currentTimeMillis();
        System.out.println("第一次 get(\"user42\") = " + get("user42") + "   耗时 " + (System.currentTimeMillis() - t0) + " ms（真去算了）");
        long t1 = System.currentTimeMillis();
        System.out.println("第二次 get(\"user42\") = " + get("user42") + "   耗时 " + (System.currentTimeMillis() - t1) + " ms（命中缓存，秒回）");
        System.out.println(">> 算过的结果存本地，第二次直接拿 → 这就是'纵向减负'里的本地缓存。");
    }
}
运行中… javac D07_LocalCache.java && java D07_LocalCache
▌ 真实输出（本机 10 核 · OpenJDK 21 实跑）
第一次 get("user42") = 600   耗时 308 ms（真去算了）
第二次 get("user42") = 600   耗时 0 ms（命中缓存，秒回）
>> 算过的结果存本地，第二次直接拿 → 这就是'纵向减负'里的本地缓存。
关键点

computeIfAbsent 是原子的"没有就算并放入、有就直接返回"。第一次 308ms（真算），第二次 0ms（命中）。这是"纵向减负"的另一半：能不算就不算。代价是一致性——本地缓存怎么和源数据同步、过期，留到缓存层专门讲。

别被流量打死

前两轴让你能扛，这一轴保证扛不住时优雅地拒绝/降级，而不是雪崩。五件套：限流、超时、熔断、隔离、降级。

D08

限流：信号量发"并发许可证"

10 个请求同时涌来，但同时只允许 3 个进入处理，其余直接拒。

import java.util.concurrent.*;
// 轴③ 横切兜底 | 限流：信号量发"并发许可证"，超额直接拒
public class D08_RateLimit {
    static final Semaphore permits = new Semaphore(3);   // 同时只放 3 个进去
    static void handle(int id) {
        if (!permits.tryAcquire()) {                     // 拿不到许可 → 直接拒，不排队
            System.out.println("请求" + id + " 被限流拒绝（已满 3 个）");
            return;
        }
        try {
            System.out.println("请求" + id + " 进入处理...");
            Thread.sleep(200);
        } catch (InterruptedException e) {
        } finally {
            permits.release();                           // 处理完归还许可
        }
    }
    public static void main(String[] a) throws Exception {
        ExecutorService p = Executors.newFixedThreadPool(10);
        CountDownLatch l = new CountDownLatch(10);
        for (int i = 1; i <= 10; i++) { int id = i; p.submit(() -> { handle(id); l.countDown(); }); }
        l.await(); p.shutdown();
        System.out.println(">> 信号量 = 并发许可证。只发 3 张，第 4 个起直接拒，保护后端不被压垮。");
    }
}

信号量 Semaphore

手里 N 张许可证，tryAcquire() 拿到才进、拿不到立刻拒（不排队、不阻塞）。它限的是并发数。另一类限速率的是令牌桶/漏桶（每秒放 N 个令牌）。限流是兜底第一招：宁可拒一部分，也不让全部一起把后端压垮。

D09

超时：绝不无限等下游

下游卡住要 2 秒，但我只给它 500ms，到点就放弃走兜底。

import java.util.concurrent.*;
// 轴③ 横切兜底 | 超时：绝不无限等下游，超时即放弃
public class D09_Timeout {
    public static void main(String[] a) {
        ExecutorService p = Executors.newSingleThreadExecutor();
        Future<String> f = p.submit(() -> { Thread.sleep(2000); return "下游结果"; }); // 下游很慢(2s)
        long t = System.currentTimeMillis();
        try {
            String r = f.get(500, TimeUnit.MILLISECONDS);   // 最多等 500ms
            System.out.println("拿到：" + r);
        } catch (TimeoutException e) {
            f.cancel(true);                                 // 放弃这个调用
            System.out.println("等了 " + (System.currentTimeMillis() - t) + " ms 还没回 → 超时放弃，走兜底");
        } catch (Exception e) {
            System.out.println("其它异常：" + e);
        }
        p.shutdownNow();
        System.out.println(">> 绝不无限等下游。超时即放弃，避免线程被拖死、连锁拖垮整个服务。");
    }
}

▌ 真实输出（本机 10 核 · OpenJDK 21 实跑）

>> 绝不无限等下游。超时即放弃，避免线程被拖死、连锁拖垮整个服务。
为什么超时是兜底的命根子

Future.get(500ms) 最多等 500ms。没有超时，一个卡死的下游会把调用线程一直占住，线程池很快被占满，整个服务跟着卡死——这就是雪崩的起点。所有跨网络调用都必须设超时。

D10

熔断 + 降级：跳闸快速失败

下游连续失败 3 次就"跳闸"，之后直接快速失败不再打它；冷却后放一个试探，好了就恢复。

// 轴③ 横切兜底 | 熔断 + 降级：连续失败就跳闸快速失败，冷却后试探恢复
public class D10_CircuitBreaker {
    enum State { CLOSED, OPEN, HALF_OPEN }
    static State state = State.CLOSED;
    static int fails = 0;
    static final int THRESHOLD = 3;       // 连续失败 3 次就跳闸
    static long openedAt = 0;
    static final long COOLDOWN = 1000;    // 跳闸后冷却 1s

    static String call(boolean downstreamHealthy) {
        if (state == State.OPEN) {        // 跳闸期间：直接快速失败，不再打下游
            if (System.currentTimeMillis() - openedAt > COOLDOWN) {
                state = State.HALF_OPEN;
                System.out.println("  [冷却结束 → HALF_OPEN，放一个试探]");
            } else {
                return fallback("断路器 OPEN，直接走降级");
            }
        }
        try {
            String r = downstream(downstreamHealthy);    // 真打下游
            if (state == State.HALF_OPEN) { state = State.CLOSED; fails = 0; System.out.println("  [试探成功 → CLOSED 恢复]"); }
            return r;
        } catch (RuntimeException e) {
            fails++;
            if (fails >= THRESHOLD || state == State.HALF_OPEN) {
                state = State.OPEN; openedAt = System.currentTimeMillis();
                System.out.println("  [失败达阈值 → OPEN，断路器跳闸]");
            }
            return fallback("下游异常，走降级");
        }
    }
    static String downstream(boolean healthy) { if (!healthy) throw new RuntimeException("boom"); return "正常结果"; }
    static String fallback(String why) { return "兜底默认值(" + why + ")"; }

    public static void main(String[] a) throws Exception {
        System.out.println("下游故障，连发 5 个请求：");
        for (int i = 1; i <= 5; i++) System.out.println("请求" + i + " state=" + state + " → " + call(false));
        System.out.println("\n等待冷却 1.1s，下游已恢复，再发 3 个：");
        Thread.sleep(1100);
        for (int i = 6; i <= 8; i++) System.out.println("请求" + i + " state=" + state + " → " + call(true));
        System.out.println(">> 连续失败就跳闸快速失败(不再拖着打死下游)，冷却后放一个试探，好了就恢复。");
    }
}

断路器 Circuit Breaker · 三态

CLOSED（正常放行）→ 连续失败达阈值 → OPEN（跳闸，直接快速失败/降级，不再拖着打死下游）→ 冷却后 HALF_OPEN（放一个试探）→ 成功则回 CLOSED。降级(fallback)=主路径不可用时返回兜底值（缓存/默认）。熔断防止"一个挂了的下游把上游也拖死"的连锁雪崩。

D11

舱壁隔离：一个慢依赖不拖垮全局

2 个慢任务(各 1s)和 1 个快任务(50ms)。共享一个线程池 vs 各用各的池。

import java.util.concurrent.*;
// 轴③ 横切兜底 | 舱壁隔离：不同依赖各用各的线程池，一个变慢不拖垮其他
public class D11_Bulkhead {
    static void task(int ms) { try { Thread.sleep(ms); } catch (InterruptedException e) {} }
    public static void main(String[] a) throws Exception {
        // ① 共享一个 2 线程池：2 个慢任务把池占满，快任务被迫排队
        ExecutorService shared = Executors.newFixedThreadPool(2);
        shared.submit(() -> task(1000));
        shared.submit(() -> task(1000));
        long t = System.currentTimeMillis();
        shared.submit(() -> task(50)).get();          // 快任务其实只要 50ms
        System.out.println("【共享池】快任务实际等了 " + (System.currentTimeMillis() - t) + " ms（被慢任务堵住！）");
        shared.shutdownNow();

        // ② 隔离：慢依赖、快依赖各用各的池，互不影响
        ExecutorService slowPool = Executors.newFixedThreadPool(2);
        ExecutorService fastPool = Executors.newFixedThreadPool(2);
        slowPool.submit(() -> task(1000));
        slowPool.submit(() -> task(1000));
        long t2 = System.currentTimeMillis();
        fastPool.submit(() -> task(50)).get();
        System.out.println("【隔离池】快任务等了 " + (System.currentTimeMillis() - t2) + " ms（不受慢依赖影响）");
        slowPool.shutdownNow(); fastPool.shutdownNow();
        System.out.println(">> 舱壁隔离：给不同依赖分独立线程池，一个依赖打满/变慢，不会拖垮其他依赖。");
    }
}

>> 舱壁隔离：给不同依赖分独立线程池，一个依赖打满/变慢，不会拖垮其他依赖。
类比 · 为什么叫"舱壁"

轮船分隔成多个水密舱，一个舱进水不会沉全船。共享池里快任务被慢任务堵到 1057ms；隔离池里快任务只用 55ms。给不同依赖分配独立线程池，一个依赖打满或变慢，烧的是它自己那个池，不会波及其他依赖。
